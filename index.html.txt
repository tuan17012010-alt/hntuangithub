    <script>
        // Cập nhật đồng hồ thời gian thực
        function updateClock() {
            const now = new Date();
            const daysOfWeek = ["Chủ Nhật", "Thứ Hai", "Thứ Ba", "Thứ Tư", "Thứ Năm", "Thứ Sáu", "Thứ Bảy"];
            const dayName = daysOfWeek[now.getDay()];
            
            const day = String(now.getDate()).padStart(2, '0');
            const month = String(now.getMonth() + 1).padStart(2, '0');
            const year = now.getFullYear();

            const hours = String(now.getHours()).padStart(2, '0');
            const minutes = String(now.getMinutes()).padStart(2, '0');
            const seconds = String(now.getSeconds()).padStart(2, '0');

            const clockString = `${dayName}, ${day}/${month}/${year} - ${hours}:${minutes}:${seconds}`;
            document.getElementById("clock-display").innerText = clockString;
        }

        setInterval(updateClock, 1000);
        updateClock();

        let activeSide = null;
        let currentRate = 0;

        // Xử lý sự kiện khi dán/nhập mã MD5 vào ô
        function handleCipherInput() {
            const inputVal = document.getElementById("cipher-input").value.trim();
            const btnTai = document.getElementById("btn-tai");
            const btnXiu = document.getElementById("btn-xiu");
            const resultBox = document.getElementById("result-text");

            if (inputVal.length > 0) {
                let cleanText = inputVal.replace(/\s+/g, '');
                let hashVal = 0;
                for (let i = 0; i < cleanText.length; i++) {
                    hashVal += cleanText.charCodeAt(i);
                }

                activeSide = (hashVal % 2 === 0) ? "Tài" : "Xỉu";
                currentRate = (hashVal % 11) + 90; // Tỉ lệ ngẫu nhiên từ 90% đến 100%

                if (activeSide === "Tài") {
                    btnTai.classList.add("active-rainbow-text");
                    btnXiu.classList.remove("active-rainbow-text");
                } else {
                    btnXiu.classList.add("active-rainbow-text");
                    btnTai.classList.remove("active-rainbow-text");
                }
                
                resultBox.innerHTML = `<span style="color: #00ffd5;">Dự đoán: <b style="color:#fff;">${activeSide.toUpperCase()}</b> | Tỉ lệ: <b style="color: #3fb950;">${currentRate}%</b></span>`;
            } else {
                resetGameUI();
            }
        }

        // Hàm xóa trắng nội dung khi bấm lại vào ô dán mã hóa
        function clearCipherInput() {
            const inputField = document.getElementById("cipher-input");
            inputField.value = ""; 
            resetGameUI();         
        }

        // Đưa giao diện về trạng thái ban đầu
        function resetGameUI() {
            activeSide = null;
            currentRate = 0;
            const btnTai = document.getElementById("btn-tai");
            const btnXiu = document.getElementById("btn-xiu");
            const resultBox = document.getElementById("result-text");

            btnTai.classList.remove("active-rainbow-text");
            btnXiu.classList.remove("active-rainbow-text");
            resultBox.innerHTML = "Dán mã MD5 ở đây để phân tích!";
        }

        // Xử lý logic kết quả khi bấm nút
        function playMD5(userChoice) {
            const inputVal = document.getElementById("cipher-input").value.trim();
            if (inputVal.length === 0) {
                document.getElementById("result-text").innerHTML = `<span style="color: #f85149;">Vui lòng dán mã MD5 vào ô bên dưới trước!</span>`;
                return;
            }

            let cleanText = inputVal.replace(/\s+/g, '') + userChoice;
            let hashVal = 0;
            for (let i = 0; i < cleanText.length; i++) {
                hashVal = (hashVal << 5) - hashVal + cleanText.charCodeAt(i);
                hashVal |= 0;
            }
            hashVal = Math.abs(hashVal);

            let dice1 = (hashVal % 6) + 1;
            let dice2 = (Math.floor(hashVal / 6) % 6) + 1;
            let dice3 = (Math.floor(hashVal / 36) % 6) + 1;
            let totalSum = dice1 + dice2 + dice3;
            
            let actualResult = (totalSum >= 11) ? "Tài" : "Xỉu";
            const resultBox = document.getElementById("result-text");
            
            const diceString = `🎲 (${dice1} - ${dice2} - ${dice3}) = <b>${totalSum} điểm</b> (${actualResult})`;

            if (userChoice === actualResult) {
                resultBox.innerHTML = `${diceString}<br><span style="color: #3fb950; font-weight: bold;">🎉 Đúng bên sáng (${currentRate}%)!</span>`;
            } else {
                resultBox.innerHTML = `${diceString}<br><span style="color: #f85149; font-weight: bold;">❌ Ngược lại với bên sáng!</span>`;
            }
        }
